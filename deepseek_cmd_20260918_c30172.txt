:: 1. Создаём корневую папку и переходим в неё
mkdir HomeLibrary
cd HomeLibrary

:: 2. Создаём решение
dotnet new sln -n HomeLibrary

:: 3. Создаём проект Minimal API
dotnet new web -n HomeLibrary

:: 4. Добавляем проект в решение
dotnet sln add HomeLibrary\HomeLibrary.csproj

:: 5. Переходим в проект
cd HomeLibrary

:: 6. Добавляем NuGet-пакеты
dotnet add package Dapper
dotnet add package Npgsql
dotnet add package Swashbuckle.AspNetCore

:: 7. Создаём папки внутри проекта
mkdir Data
mkdir Models
mkdir Repositories
mkdir Services
mkdir wwwroot
mkdir wwwroot\css
mkdir wwwroot\js

:: 8. Создаём пустые файлы-заготовки внутри проекта
type nul > Data\DbContext.cs
type nul > Models\Book.cs
type nul > Models\BookCreateDto.cs
type nul > Repositories\BookRepository.cs
type nul > Services\BookService.cs
type nul > wwwroot\index.html
type nul > wwwroot\book.html
type nul > wwwroot\css\site.css
type nul > wwwroot\js\api.js
type nul > wwwroot\js\list.js
type nul > wwwroot\js\card.js

:: 9. Возвращаемся в корень решения
cd ..

:: 10. Создаём папку для БД и файлы инфраструктуры
mkdir db
type nul > db\init.sql
type nul > Dockerfile
type nul > docker-compose.yml
type nul > .dockerignore
type nul > .gitignore
type nul > README.md
